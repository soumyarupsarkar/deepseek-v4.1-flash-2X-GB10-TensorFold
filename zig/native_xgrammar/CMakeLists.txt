cmake_minimum_required(VERSION 3.18)
project(tensorfold_xgrammar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT XGRAMMAR_SOURCE_DIR)
  message(FATAL_ERROR "Set XGRAMMAR_SOURCE_DIR to the verified xgrammar v0.2.8 checkout (97787376faee5ed8466cfad57c99855e4ce2f6aa)")
endif()
set(XGRAMMAR_BUILD_PYTHON_BINDINGS OFF CACHE BOOL "" FORCE)
set(XGRAMMAR_BUILD_CXX_TESTS OFF CACHE BOOL "" FORCE)
set(XGRAMMAR_ENABLE_CPPTRACE OFF CACHE BOOL "" FORCE)
# The pinned dependency includes cmake/config.cmake, whose normal variable
# enables Python bindings even over a forced cache value. Its build-directory
# config has precedence. Keep this dedicated native bridge independent of
# Python development headers and TVM's Python binding configuration.
configure_file(native_config.cmake ${CMAKE_BINARY_DIR}/config.cmake COPYONLY)
add_subdirectory(${XGRAMMAR_SOURCE_DIR} xgrammar-build EXCLUDE_FROM_ALL)
add_library(tf_xgrammar SHARED bridge.cc)
target_link_libraries(tf_xgrammar PRIVATE xgrammar)
target_include_directories(tf_xgrammar SYSTEM PRIVATE ${XGRAMMAR_SOURCE_DIR}/3rdparty/picojson)
target_compile_options(tf_xgrammar PRIVATE -Wall -Wextra -Werror)
set_target_properties(tf_xgrammar PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
install(TARGETS tf_xgrammar LIBRARY DESTINATION lib COMPONENT native)
enable_testing()
add_executable(tf_xgrammar_test bridge_test.cc)
target_link_libraries(tf_xgrammar_test PRIVATE tf_xgrammar)
target_include_directories(tf_xgrammar_test SYSTEM PRIVATE ${XGRAMMAR_SOURCE_DIR}/3rdparty/picojson)
target_compile_options(tf_xgrammar_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME native_grammar COMMAND tf_xgrammar_test)
